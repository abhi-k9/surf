cmake_minimum_required(VERSION 3.21)
project(NPMS VERSION 0.1.0 LANGUAGES CXX)
option(NPMS_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(NPMS_PYTHON_TESTS "Run the independent Python/NumPy oracle if installed" ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(npms src/core.cpp)
target_include_directories(npms PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(npms PUBLIC cxx_std_20)
add_executable(npms_cli src/main.cpp)
target_link_libraries(npms_cli PRIVATE npms)
include(CTest)
if(BUILD_TESTING)
  add_executable(npms_tests tests/test_core.cpp)
  target_link_libraries(npms_tests PRIVATE npms)
  add_test(NAME test_fixture_contract COMMAND npms_tests ${CMAKE_CURRENT_SOURCE_DIR}/handoff)
endif()
foreach(target npms npms_cli npms_tests)
  if(TARGET ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
      if(NPMS_SANITIZE)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
      endif()
    endif()
  endif()
endforeach()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(BUILD_TESTING AND Python3_FOUND)
  add_test(NAME test_worker_security COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_worker.py $<TARGET_FILE:npms_cli>)
  if(NPMS_PYTHON_TESTS)
    execute_process(COMMAND ${Python3_EXECUTABLE} -c "import numpy" RESULT_VARIABLE HAVE_NUMPY OUTPUT_QUIET ERROR_QUIET)
    if(HAVE_NUMPY EQUAL 0)
      add_test(NAME test_python_oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_oracle.py $<TARGET_FILE:npms_cli>)
      set_tests_properties(test_python_oracle PROPERTIES ENVIRONMENT "OPENBLAS_NUM_THREADS=1")
    else()
      message(STATUS "NumPy unavailable: optional independent oracle test not registered")
    endif()
  endif()
endif()
